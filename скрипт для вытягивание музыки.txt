// Зайти на плейлист, прогрузить его, F12, консоль, вставить то что ниже

function exportPlaylist() {
    const songs = [];
    // ИСПРАВЛЕНО: ytmusic вместо yt-music
    const rows = document.querySelectorAll('ytmusic-responsive-list-item-renderer');
    
    rows.forEach(row => {
        // Название песни
        const titleEl = row.querySelector('.title-column yt-formatted-string');
        let title = "Unknown";
        if (titleEl) {
            // Иногда title лежит в атрибуте title, иногда в тексте
            title = titleEl.getAttribute('title') || titleEl.innerText;
        }
        
        // Исполнитель
        // Находится во второй колонке. Там может быть: "Исполнитель • Альбом • Время"
        const secondaryEl = row.querySelector('.secondary-flex-columns yt-formatted-string');
        let artist = "Unknown";
        
        if (secondaryEl) {
            // Пытаемся найти ссылку на артиста (обычно она первая)
            const artistLink = secondaryEl.querySelector('a');
            if (artistLink) {
                artist = artistLink.innerText;
            } else {
                // Если ссылок нет (редко, но бывает), берем текст до первой точки
                // Например: "Кишлак • Сборник" -> берем "Кишлак"
                const text = secondaryEl.innerText;
                artist = text.split(' • ')[0] || text;
            }
        }

        if (title !== "Unknown") {
            // Очистка от лишних пробелов
            songs.push({ 
                title: title.trim(), 
                artist: artist.trim() 
            });
        }
    });

    console.log(`✅ Найдено треков: ${songs.length}`);
    
    if (songs.length > 0) {
        // Копируем в буфер
        copy(`const musicData = ${JSON.stringify(songs, null, 4)};`);
        console.log("📋 Данные скопированы в буфер обмена! Создай файл data/music-data.js и вставь их туда (Ctrl+V).");
    } else {
        console.log("❌ Треки не найдены. Убедись, что ты на странице Плейлиста или 'Понравившиеся' (Liked Music).");
    }
}

exportPlaylist();
